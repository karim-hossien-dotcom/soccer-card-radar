# Soccer Card Radar v2 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rebuild the card scanner so research drives the search pipeline — every listing has a reason, no junk gets in, and Karim can decide "buy or skip" in under 10 seconds.

**Architecture:** Research engine defines 3 lanes (Rookie Scout, Current Steals, Legacy). Each lane produces research actions with exact search queries. Scan pipeline searches eBay sold for comps, then eBay active for deals, stores only cards priced below market. Dashboard has 3 tabs: Research (the brain), Buy (act on deals), Sell/Avoid (overpriced).

**Tech Stack:** Next.js 14, Supabase, eBay Browse API, Recharts, web search for stats

```typescript
/**
 * Quality filter — applied at search time, not display time.
 * If a card doesn't pass this, it never enters the DB.
 */
export function isQualityCard(title: string): boolean {
  const t = title.toLowerCase();

  // HARD REJECT
  if (t.includes("leaf")) return false;
  if (t.includes("hobby box") || t.includes("sealed") || t.includes("blaster") || t.includes("mega box") || t.includes("pack")) return false;
  if (t.includes(" lot ") || t.startsWith("lot ") || t.includes("lot of")) return false;
  if (t.includes("sticker") || t.includes("album") || t.includes("jersey") || t.includes("shirt") || t.includes("figurine")) return false;
  if (t.includes("base") && !t.includes("auto")) return false;
  if (t.includes("singles") || t.includes("pick your") || t.includes("you pick")) return false;
  if (t.includes("digital") || t.includes("topps bunt")) return false;
  if (t.includes("sticker auto")) return false;

  // MUST HAVE value marker
  const numMatch = t.match(/\/(\d{1,2})\b/);
  const hasSmallNumber = numMatch && parseInt(numMatch[1]) <= 99;
  const isAuto = (t.includes("auto") || t.includes("autograph") || t.includes("signature")) && !t.includes("sticker auto");
  const isGraded = t.includes("psa") || t.includes("bgs") || t.includes("sgc") || t.includes("cgc");
  const isSSP = t.includes("ssp") || t.includes("case hit") || t.includes("1/1");

  return hasSmallNumber || isAuto || isGraded || isSSP;
}

/**
 * Exception for pre-2015 legend cards where base can be valuable.
 */
export function isVintageLegend(title: string): boolean {
  const t = title.toLowerCase();
  const legends = ["messi", "ronaldo", "maradona", "pele", "zidane"];
  const isLegend = legends.some((l) => t.includes(l));
  if (!isLegend) return false;

  const yearMatch = t.match(/\b(19|200|201[0-4])\d?\b/);
  return !!yearMatch;
}
```
